# llama.cpp bindings. PyPI publishes source-only, so the prebuilt CUDA wheels come
# from the index below. Those wheels are `py3-none-<platform>` — ABI-agnostic,
# because llama.cpp ships as a shared library loaded via ctypes rather than as a
# CPython extension, so any Python 3.x works.
#
# cu125 is the newest CUDA variant published there. It drives Blackwell (sm_120)
# correctly via PTX JIT — verified on an RTX 5060 Ti — so there is no need to
# compile from source or install the CUDA toolkit.
--extra-index-url https://abetlen.github.io/llama-cpp-python/whl/cu125
llama-cpp-python==0.3.34

# The wheel bundles ggml-cuda.dll but NOT the CUDA runtime it links against.
# torch used to satisfy that by accident; with torch gone these are required, and
# app/modules/llm/cuda_bootstrap.py preloads them before llama_cpp is imported.
nvidia-cuda-runtime-cu12
nvidia-cublas-cu12

# Chat templates ship inside the GGUF, so jinja2 renders them directly — no
# transformers, no tokenizer download.
jinja2
huggingface-hub

# Service / API
fastapi>=0.95.0
uvicorn[standard]>=0.22.0

# Image module (Real-ESRGAN upscaler). PyPI torch is CPU-only on Windows, which is
# what the upscaler runs on (the GPU is held by the LLM). The net is a small plain
# PyTorch module, so no basicsr/realesrgan packages are needed.
torch
numpy
opencv-python-headless
httpx
